def scalar(value, preferred=()):
    if isinstance(value, (int, float, np.integer, np.floating)) and not isinstance(value, bool):
        return float(value)
    if isinstance(value, dict):
        for k in preferred:
            if k in value:
                x = scalar(value[k], preferred)
                if x is not None: return x
        for k, v in value.items():
            if str(k).lower() in {'percent','percentage','pct','score','value','mean'}:
                x = scalar(v, preferred)
                if x is not None: return x
    return None
